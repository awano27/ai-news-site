📝 サンプルコード実装

# Python SDK使用例（ドキュメントに基づく想定）
import requests

# API設定
api_key = 'your_api_key'  # サイトから取得
api_endpoint = 'https://api.aicaretoc.io/rx/analyze'

# 患者データ準備
patient_data = {
    'patient_info': {
        'age': 30,
        'genetics': 'sample_data',
        'health_history': ['diabetes', 'hypertension'],
        'lifestyle': {
            'exercise': 'moderate',
            'diet': 'balanced'
        }
    }
}

# API呼び出し
response = requests.post(
    api_endpoint, 
    json=patient_data, 
    headers={'Authorization': api_key}
)

# 結果表示
result = response.json()
print(f"推奨栄養素: {result['recommendations']}")
print(f"サプリメント提案: {result['supplements']}")
